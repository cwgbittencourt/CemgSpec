        finally
        {

        }

    private static bool VerifyOnlyRestriction(IntPtr token,IntPtr expected,IntPtr logon)
    {var data=ReadToken(token,11);try{if(Marshal.ReadInt32(data)!=2)return false;var first=Marshal.ReadIntPtr(data,8);var second=Marshal.ReadIntPtr(data,8+Marshal.SizeOf<SidAttributes>());return (EqualSid(first,expected)&&EqualSid(second,logon))||(EqualSid(first,logon)&&EqualSid(second,expected));}finally{Marshal.FreeHGlobal(data);}}
    private static bool IsGroupEnabled(IntPtr token,IntPtr sid)
    {var data=ReadToken(token,2);try{for(var i=0;i<Marshal.ReadInt32(data);i++){var entry=Marshal.PtrToStructure<SidAttributes>(data+8+i*Marshal.SizeOf<SidAttributes>());if(EqualSid(entry.Sid,sid))return(entry.Attributes&4)!=0;}return false;}finally{Marshal.FreeHGlobal(data);}}
    private static int ReadCount(IntPtr token,int kind){var data=ReadToken(token,kind);try{return Marshal.ReadInt32(data);}finally{Marshal.FreeHGlobal(data);}}
    private static IntPtr ReadToken(IntPtr token,int kind){GetTokenInformation(token,kind,IntPtr.Zero,0,out var needed);if(needed<4)throw new Win32Exception();var data=Marshal.AllocHGlobal(needed);if(!GetTokenInformation(token,kind,data,needed,out needed)){Marshal.FreeHGlobal(data);throw new Win32Exception();}return data;}
