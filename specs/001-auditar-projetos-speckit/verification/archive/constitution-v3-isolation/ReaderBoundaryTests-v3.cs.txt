using SpecAudit.Domain;
using SpecAudit.Infrastructure.Documents;
using SpecAudit.Windows.Tests.Fixtures;
namespace SpecAudit.Windows.Tests;
public sealed class ReaderBoundaryTests
{
    [Fact]
    public void CatalogReadsOnlyRecognizedDocumentsAndRejectsForeignIds()
    {
        var fixture=DocumentaryFixture.Create();var project=Guid.NewGuid();
        using var catalog=new DocumentCatalog(project,fixture.Origin);
        var entry=Assert.Single(catalog.Entries,e=>e.RelativePath.Replace('\\','/')=="specs/001-demo/spec.md");
        Assert.DoesNotContain(catalog.Entries,e=>e.RelativePath.EndsWith(".cs"));
        var broker=new DocumentBroker(catalog);
        var snapshot=broker.Read(project,entry.DocumentId,0);
        Assert.Contains("Demo feature",System.Text.Encoding.UTF8.GetString(snapshot.CopyBytes()));
        Assert.Equal("OutOfScope",Assert.Throws<DocumentAccessException>(()=>broker.Read(Guid.NewGuid(),entry.DocumentId,0)).Code);
        Assert.Equal("OutOfScope",Assert.Throws<DocumentAccessException>(()=>broker.Read(project,Guid.NewGuid(),0)).Code);
        Assert.True(fixture.IsUnchanged());
    }
    [Fact]
    public void RejectsJunctionAtActualOpenBeforeReadingAnyBytes()
    {
        var fixture=DocumentaryFixture.Create();var project=Guid.NewGuid();
        using var catalog=new DocumentCatalog(project,fixture.Origin);
        var entry=Assert.Single(catalog.Entries,e=>e.RelativePath.Replace('\\','/')=="specs/001-demo/spec.md");
        var feature=fixture.Feature;var parked=Path.Combine(fixture.Root,"parked");
        Directory.Move(feature,parked);
        FixtureJunction.Create(fixture.Root,feature,fixture.Output);
        File.WriteAllText(Path.Combine(fixture.Output,"spec.md"),"unapproved target");
        var bytesRead=false;var broker=new DocumentBroker(catalog){BeforeRead=()=>bytesRead=true};
        var failure=Assert.Throws<DocumentAccessException>(()=>broker.Read(project,entry.DocumentId,0));
        Assert.Equal("OutOfScope",failure.Code);Assert.False(bytesRead);
    }
    [Fact]
    public void ReplacedFileAndChangedGenerationDoNotReturnSnapshot()
    {
        var fixture=DocumentaryFixture.Create();var project=Guid.NewGuid();
        using var catalog=new DocumentCatalog(project,fixture.Origin);
        var entry=Assert.Single(catalog.Entries,e=>e.RelativePath.Replace('\\','/')=="specs/001-demo/spec.md");
        var path=Path.Combine(fixture.Feature,"spec.md");
        File.Move(path,Path.Combine(fixture.Feature,"old.md"));File.WriteAllText(path,"replacement");
        var broker=new DocumentBroker(catalog);
        Assert.Equal("Changed",Assert.Throws<DocumentAccessException>(()=>broker.Read(project,entry.DocumentId,0)).Code);
        Assert.Equal("Changed",Assert.Throws<DocumentAccessException>(()=>broker.Read(project,entry.DocumentId,1)).Code);
    }

    [Fact]
    public async Task ExecutorAndChildCannotReadCodeConfigOrMutateSyntheticOrigin()
    {
        var fixture=DocumentaryFixture.Create();File.WriteAllText(Path.Combine(fixture.Origin,"appsettings.json"),"{\"synthetic\":true}");fixture.CaptureBaseline();
        var checkout=new DirectoryInfo(AppContext.BaseDirectory);
        while(checkout is not null && !File.Exists(Path.Combine(checkout.FullName,"SpecAudit.slnx")))checkout=checkout.Parent;
        var executable=Path.Combine(checkout!.FullName,"tests","SpecAudit.Windows.Tests","bin","native","Worker.exe");
        var worker=new SpecAudit.Infrastructure.Execution.TrustedWorker(executable,DocumentaryFixture.Hash(executable),fixture.Origin);
        using var channel=new SpecAudit.Infrastructure.Execution.ExecutionChannel(Guid.NewGuid(),0);
        var result=await new SpecAudit.Infrastructure.Execution.RestrictedLauncher().RunAsync(worker,channel,10000);
        File.WriteAllText(Path.Combine(checkout.FullName,"specs","001-auditar-projetos-speckit","evidence","logs","reader-process-proof.json"),System.Text.Json.JsonSerializer.Serialize(result,new System.Text.Json.JsonSerializerOptions{WriteIndented=true}));
        Assert.Equal(0,result.ExitCode);Assert.Null(result.Error);Assert.True(result.IsAppContainer);Assert.False(result.AdminEnabled);
        Assert.Equal(2,result.Output.Split("READ_CODE_CONFIG_WRITE_CREATE_RENAME_DELETE_DENIED",StringSplitOptions.None).Length-1);
        Assert.Equal(0u,result.ActiveAfterTermination);Assert.Equal(0,result.ProfileDeleteResult);Assert.True(fixture.IsUnchanged());
        using var catalog=new DocumentCatalog(Guid.NewGuid(),fixture.Origin);
        var spec=Assert.Single(catalog.Entries,e=>e.RelativePath.Replace('\\','/')=="specs/001-demo/spec.md");
        var broker=new DocumentBroker(catalog){AccessObserved=fixture.RecordAccess};
        Assert.NotEmpty(broker.Read(catalog.ProjectId,spec.DocumentId,0).CopyBytes());
        File.AppendAllText(Path.Combine(fixture.Feature,"spec.md"),"\nUser edit is still allowed.");
        Assert.Contains("User edit",File.ReadAllText(Path.Combine(fixture.Feature,"spec.md")));
        Assert.Equal("Changed",Assert.Throws<DocumentAccessException>(()=>broker.Read(catalog.ProjectId,spec.DocumentId,0)).Code);
    }

    [Fact]
    public async Task BoundaryMustAlsoDenyCodeWhenOriginHasBroadAppPackageReadPermission()
    {
        // Synthetic fixture setup only. Never change the ACL of a user's solution.
        if(!OperatingSystem.IsWindows())throw new PlatformNotSupportedException();
        var fixture=DocumentaryFixture.Create();File.WriteAllText(Path.Combine(fixture.Origin,"appsettings.json"),"{}");
        var directory=new DirectoryInfo(fixture.Origin);var acl=System.IO.FileSystemAclExtensions.GetAccessControl(directory);
        acl.AddAccessRule(new System.Security.AccessControl.FileSystemAccessRule(new System.Security.Principal.SecurityIdentifier("S-1-15-2-1"),System.Security.AccessControl.FileSystemRights.ReadAndExecute,System.Security.AccessControl.InheritanceFlags.ContainerInherit|System.Security.AccessControl.InheritanceFlags.ObjectInherit,System.Security.AccessControl.PropagationFlags.None,System.Security.AccessControl.AccessControlType.Allow));
        System.IO.FileSystemAclExtensions.SetAccessControl(directory,acl);fixture.CaptureBaseline();
        var checkout=new DirectoryInfo(AppContext.BaseDirectory);
        while(checkout is not null && !File.Exists(Path.Combine(checkout.FullName,"SpecAudit.slnx")))checkout=checkout.Parent;
        var executable=Path.Combine(checkout!.FullName,"tests","SpecAudit.Windows.Tests","bin","native","Worker.exe");
        var worker=new SpecAudit.Infrastructure.Execution.TrustedWorker(executable,DocumentaryFixture.Hash(executable),fixture.Origin);
        using var channel=new SpecAudit.Infrastructure.Execution.ExecutionChannel(Guid.NewGuid(),0);
        var result=await new SpecAudit.Infrastructure.Execution.RestrictedLauncher().RunAsync(worker,channel,10000);
        File.WriteAllText(Path.Combine(checkout.FullName,"specs","001-auditar-projetos-speckit","evidence","logs","reader-broad-acl-proof.json"),System.Text.Json.JsonSerializer.Serialize(result,new System.Text.Json.JsonSerializerOptions{WriteIndented=true}));
        Assert.Equal(0,result.ExitCode);Assert.Null(result.Error);Assert.True(fixture.IsUnchanged());
    }
}
